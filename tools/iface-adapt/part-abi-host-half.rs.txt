/// `guard` on caller-owned, fixed output storage: the same owner and watchdog word; the loan cannot
/// grow, escape the callback or transfer the lender's allocation. None: busbar-ffi's ACTIVE stack is full.
pub fn guard_fixed<R>(
    slot: u32,
    entry: u8,
    out: &mut [u8],
    call: impl for<'loan> FnOnce(&mut busbar_ffi::host::OutLease<'loan>) -> R,
) -> Option<R> {
    crate::host_side::driver::step_fixed(slot, entry, out, call)
}

// 4. driver.rs, beside `set_fault_sink`: the resume tap of the crossing recorder. The results a parked
//    chain is resumed with are bytes the engine hands a plugin, so the recorder must see them.

/// Installs the process's resume tap: `tap(slot, results)` runs just before a parked chain of `slot` is
/// resumed with `results`. A second call is ignored. With none installed a resume pays one load.
pub fn set_resume_tap(tap: fn(slot: u32, results: &[u8])) {
    let _ = RESUME_TAP.set(tap);
}

static RESUME_TAP: OnceLock<fn(u32, &[u8])> = OnceLock::new();

#[inline(always)]
pub(crate) fn resume_tap(slot: u32, results: &[u8]) {
    if let Some(tap) = RESUME_TAP.get() {
        tap(slot, results);
    }
}

// 5. watch.rs: the hang report names the thread that holds the hung entry.

// (a) beside `struct Table`:

/// The thread behind one watch word, recorded when the thread claims the word (once per thread, cold)
/// and handed to the scan's report with the word it found stuck.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Held {
    /// The thread's name; None for a thread started without one.
    pub thread_name: Option<Box<str>>,
    /// The thread's id, as `std::thread::ThreadId::as_u64` numbers it.
    pub thread_id: u64,
    /// The index of the thread's watch word.
    pub word: u32,
}

// (b) in `struct Table`, one more field, allocated in `table()` with `WATCH_CAP` empty entries
//     (`(0..WATCH_CAP).map(|_| Mutex::new(None)).collect()`):
//
//        /// Who claimed each word (written at claim, cleared at release; read by the scanner's report).
//        holders: Box<[Mutex<Option<Held>>]>,
//
// (c) in `claim`, in the `Ok(_)` arm, after `claim.idx.set(idx)` (cold: the thread's first entry):

fn record_holder(table: &Table, idx: u32) {
    let thread = std::thread::current();
    if let Some(holder) = table.holders.get(idx as usize) {
        *holder.lock().unwrap_or_else(PoisonError::into_inner) = Some(Held {
            thread_name: thread.name().map(Box::from),
            thread_id: thread.id().as_u64().get(),
            word: idx,
        });
    }
}

// (d) in the thread-exit release of a word (the TLS destructor that clears the bitmap bit), clear the
//     holder the same way (`*holder.lock()... = None`).
//
// (e) `scan`'s report gains the holder. Signature:
//
//        mut report: impl FnMut(u32 /*slot*/, u32 /*epoch*/, u8 /*entry*/, u64 /*busy_ms*/, &Held),
//
//     and at the one `report(...)` call, with `i` the word index:

fn report_with_holder(
    table: &Table,
    i: usize,
    report: &mut impl FnMut(u32, u32, u8, u64, &Held),
    (slot, epoch, entry, busy_ms): (u32, u32, u8, u64),
) {
    let unknown = Held {
        thread_name: None,
        thread_id: 0,
        word: i as u32,
    };
    let held = table
        .holders
        .get(i)
        .and_then(|holder| holder.lock().unwrap_or_else(PoisonError::into_inner).clone())
        .unwrap_or(unknown);
    report(slot, epoch, entry, busy_ms, &held);
}
